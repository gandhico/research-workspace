\section*{Conventions}

\begin{itemize}[leftmargin=1.4em,itemsep=1pt]
\item $\mu[n]$ is the unit step, $\delta[n]$ the unit sample.
\item $\circledast$ denotes \emph{linear} convolution,
      $\displaystyle (x\circledast h)[n]=\sum_{k=-\infty}^{\infty}x[k]\,h[n-k]$.
\item Energy $\mathcal{E}_x=\sum_n |x[n]|^2$; average power
      $\mathcal{P}_x=\lim\limits_{K\to\infty}\dfrac{1}{2K+1}\sum_{n=-K}^{K}|x[n]|^2$.
\item Correlation: $r_{xy}[\ell]=\sum_n x[n]\,y[n-\ell]$, \ $r_{x}[\ell]=r_{xx}[\ell]$.
\item Conjugate-symmetric / antisymmetric parts:
      $x_{\mathrm{cs}}[n]=\tfrac12\bigl(x[n]+x^*[-n]\bigr)$,
      $x_{\mathrm{ca}}[n]=\tfrac12\bigl(x[n]-x^*[-n]\bigr)$;
      even / odd parts $x_{\mathrm{ev}},x_{\mathrm{od}}$ are the real-sequence specialization.
\item The sequences of \textbf{Problem 2.3}, used repeatedly below, are
\[
\begin{aligned}
\{x[n]\}&=\{3,\,-4,\,-2,\,5,\,0,\,1,\,-3\}, && -1\le n\le 5,\\
\{y[n]\}&=\{-5,\,-1,\,0,\,3,\,6,\,-2,\,8\}, && -3\le n\le 3,\\
\{w[n]\}&=\{-1,\,3,\,-4,\,0,\,1,\,-5,\,6\}, && \phantom{-}3\le n\le 9.
\end{aligned}
\]
\end{itemize}
