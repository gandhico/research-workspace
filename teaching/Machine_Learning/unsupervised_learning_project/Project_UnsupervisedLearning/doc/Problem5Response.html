Problem 5: Online K-means

Part a)

myKmeansOnline shuffles the observations at the beginning of every epoch. For each observation it finds the closest centroid and applies

\[ \boldsymbol{\mu}_{I}^{\mathrm{new}} =\boldsymbol{\mu}_{I}^{\mathrm{old}} +\eta\left(\mathbf{x}_n-\boldsymbol{\mu}_{I}^{\mathrm{old}}\right). \]

After an epoch, the maximum Euclidean movement among all centroids is compared with the fixed tolerance. The loop stops on convergence or at the maximum of 100 epochs.

Random.seed!(5101)

# One update with eta = 0.5 must move the center halfway to the observation.
one_point_center, one_point_epochs = myKmeansOnline(
    [[2.0, 2.0]],
    [[0.0, 0.0]],
    0.5;
    maxIter=1,
    tol=0.0,
    verbose=false,
)
@assert one_point_center ≈ [[1.0, 1.0]]
@assert one_point_epochs == 1

# Centers already located on two degenerate clusters should converge in one epoch.
online_test_data = [[0.0, 0.0], [5.0, 5.0], [0.0, 0.0], [5.0, 5.0]]
online_test_centers, online_test_epochs = myKmeansOnline(
    online_test_data,
    [[0.0, 0.0], [5.0, 5.0]],
    0.1;
    maxIter=100,
    tol=1e-12,
    verbose=false,
)
@assert online_test_centers ≈ [[0.0, 0.0], [5.0, 5.0]]
@assert online_test_epochs == 1
println("Problem 5 online K-means validation passed.")

Part b)

The convergence tolerance is fixed at $3\times10^{-4}$. Twenty-five learning rates are spaced logarithmically from $10^{-6}$ through $10^0$. For every learning rate, 30 independent data sets and random initializations are tested. A run that reaches the 100-epoch safety limit contributes 100 to the average.

Random.seed!(5102)
eta_values = 10.0 .^ range(-6, 0; length=25)
online_tolerance = 3e-4
online_max_epochs = 100
online_trials = 30
average_epochs = Float64[]
convergence_rates = Float64[]

for eta in eta_values
    epochs_for_eta = Int[]
    for _ in 1:online_trials
        data = online_two_gaussian_data(500)
        initial = online_random_unique_centers(data, 2)
        _, epochs = myKmeansOnline(
            data,
            initial,
            eta;
            maxIter=online_max_epochs,
            tol=online_tolerance,
            verbose=false,
        )
        push!(epochs_for_eta, epochs)
    end
    push!(average_epochs, mean(epochs_for_eta))
    push!(convergence_rates, count(<(online_max_epochs), epochs_for_eta) / online_trials)
end

best_eta_index = argmin(average_epochs)
best_eta = eta_values[best_eta_index]
best_average_epochs = average_epochs[best_eta_index]

println("Selected learning rate eta: ", best_eta)
println("Average epochs at selected eta: ", round(best_average_epochs, digits=2))
println(
    "Fraction finishing before the epoch cap: ",
    round(convergence_rates[best_eta_index], digits=2),
)

plot(
    eta_values,
    average_epochs;
    marker=:circle,
    linewidth=2,
    xscale=:log10,
    xlabel="Learning rate, eta",
    ylabel="Average epochs",
    title="Online K-means learning-rate search (30 trials)",
    label="average epochs",
    legend=:bottomright,
)
scatter!(
    [best_eta],
    [best_average_epochs];
    markersize=8,
    marker=:star5,
    label="selected eta",
)

For this fixed tolerance and the deterministic experiment seed, the selected learning rate is printed above the figure. Very small rates make too little progress, while larger rates continue to move the centers as the shuffled observations arrive. The intermediate selected rate gives the smallest capped average. The convergence fraction is reported with the average because an epoch count of 100 can mean that the safety limit, rather than the movement threshold, ended the run.
